cmake_minimum_required(VERSION 3.24)
project(AifEmbeddedConsumer LANGUAGES C CXX)
get_filename_component(AIF_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
set(HWY_ENABLE_INSTALL OFF)
set(HWY_ENABLE_TESTS OFF)
set(HWY_ENABLE_CONTRIB OFF)
set(HWY_ENABLE_EXAMPLES OFF)
set(HWY_ENABLE_TOOLS OFF)
set(HWY_LIBRARY_TYPE STATIC)
add_subdirectory("${AIF_SOURCE}/third_party/highway" hwy)
add_subdirectory("${AIF_SOURCE}" aif)
if(AIF_ENABLE_INSTALL)
  message(FATAL_ERROR "Embedded builds must default to no install rules")
endif()
add_executable(aif_embedded_consumer ../installed/consumer.c)
target_link_libraries(aif_embedded_consumer PRIVATE AvsFilters::FocusKernels)
enable_testing()
add_test(NAME embedded.consumer COMMAND aif_embedded_consumer)
