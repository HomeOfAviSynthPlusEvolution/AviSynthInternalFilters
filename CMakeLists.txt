function(aif_target_defaults target)
  target_compile_features(${target} PUBLIC cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND
     CMAKE_CXX_COMPILER_ARCHITECTURE_ID MATCHES "^ARM64" AND
     target MATCHES "^aif_(rotation|planes)_kernels$")
    # Highway cache_control.h includes arm_acle.h, unavailable in native MSVC
    # ARM64. Disable cache hints/streaming helpers, not the NEON computations.
    target_compile_definitions(${target} PRIVATE HWY_DISABLE_CACHE_CONTROL)
  endif()
  if(target MATCHES "^aif_.*_filters$")
    target_include_directories(${target} PUBLIC
      $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/src/common/include>)
    if(AIF_BUILD_AVSCORE)
      target_compile_definitions(${target} PRIVATE BUILDING_AVSCORE)
      if(NOT BUILD_SHARED_LIBS)
        target_compile_definitions(${target} PRIVATE AVS_STATIC_LIB)
      endif()
    endif()
  endif()
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      # clang-cl can contract scalar tails when /arch:AVX2 permits FMA.
      target_compile_options(${target} PRIVATE /clang:-ffp-contract=off)
    else()
      target_compile_options(${target} PRIVATE /fp:precise)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -ffp-contract=off)
  endif()
  if(AIF_SANITIZERS)
    if(MSVC)
      message(FATAL_ERROR "AIF_SANITIZERS requires GCC or Clang with GNU-style flags")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endfunction()

    HWY_WANT_AVX3_ZEN4=1)

    target_compile_definitions(hwy PRIVATE HWY_COMPILE_ONLY_SCALAR)
  endif()
endif()
